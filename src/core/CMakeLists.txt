# Core, Qt Sql (SQLite, for the clipboard history) and Win32, so it can be
# unit-tested without a GUI.

    AppCatalog.cpp AppCatalog.h
    AppLogo.cpp AppLogo.h
    ChangeWatcher.cpp ChangeWatcher.h
    ClipStore.cpp ClipStore.h
    ContentIndex.cpp ContentIndex.h
    ContentIndexer.cpp ContentIndexer.h
    ContentScanner.cpp ContentScanner.h
    Crawler.cpp Crawler.h
    DocExtractor.cpp DocExtractor.h
    Documents.h
    DoubleClickDetector.h
    DoubleTapDetector.h
    Favorites.cpp Favorites.h
    FileIndex.cpp FileIndex.h
    FileText.cpp FileText.h
    GameGuard.h
    History.cpp History.h
    IndexFolder.cpp IndexFolder.h
    IndexService.cpp IndexService.h
    MenuHotkey.h
    NameSearch.cpp NameSearch.h
    Ntfs.cpp Ntfs.h
    NtfsBulk.cpp NtfsBulk.h
    NtfsIndexer.cpp NtfsIndexer.h
    PathText.cpp PathText.h
    Pinyin.cpp Pinyin.h
    PinyinTable.cpp # generated by tools/make_pinyin.py
    Query.cpp Query.h
    Release.cpp Release.h
    SearchEngine.cpp SearchEngine.h
    SearchTypes.h
    ServiceVolumeAccess.cpp ServiceVolumeAccess.h
    SettingsMatch.cpp SettingsMatch.h
    ShareIndexer.cpp ShareIndexer.h
    ShareLister.cpp ShareLister.h
    Snapshot.cpp Snapshot.h
    SystemCatalog.cpp SystemCatalog.h
    TextUtil.cpp TextUtil.h
    Unc.cpp Unc.h
    VolumeProtocol.cpp VolumeProtocol.h
    WebShortcut.cpp WebShortcut.h
    Win32Util.cpp Win32Util.h
    WorkerPool.cpp WorkerPool.h
    Wtf8.cpp Wtf8.h
)

# WinShun's own places and keywords, on top of Windows' Settings index (SystemCatalog).
qt_add_resources(wscore "places" PREFIX "/winshun" FILES places.txt)

target_include_directories(wscore PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(wscore
    PUBLIC Qt6::Core Qt6::Sql winshun_options wsextract_api
    PRIVATE kernel32 advapi32 bcrypt cabinet cfgmgr32 netapi32 ole32 shell32 shlwapi user32 ws2_32)
